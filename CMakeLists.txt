cmake_minimum_required(VERSION 3.16)
project(VoxelWorld LANGUAGES CXX)

option(VOXEL_SYSTEM_BULLET "Use an installed Bullet SDK instead of the pinned Bullet3 source" OFF)
if(VOXEL_SYSTEM_BULLET)
    find_package(Bullet REQUIRED)
else()
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    FetchContent_Declare(bullet
        URL https://deb.debian.org/debian/pool/main/b/bullet/bullet_3.24+dfsg.orig.tar.xz
        URL_HASH SHA256=43c1edae2024ba18bb5d7bfd3bb2394b066a79151a29789504a2d8efcdc333d7)
    FetchContent_GetProperties(bullet)
    if(NOT bullet_POPULATED)
        FetchContent_Populate(bullet)
    endif()
    set(BULLET_PHYSICS_SOURCE_DIR "${bullet_SOURCE_DIR}")
    set(BULLET_VERSION 3.24)
    add_subdirectory("${bullet_SOURCE_DIR}/src/LinearMath" "${bullet_BINARY_DIR}/LinearMath")
    add_subdirectory("${bullet_SOURCE_DIR}/src/BulletCollision" "${bullet_BINARY_DIR}/BulletCollision")
    add_subdirectory("${bullet_SOURCE_DIR}/src/BulletDynamics" "${bullet_BINARY_DIR}/BulletDynamics")
    set(BULLET_INCLUDE_DIRS "${bullet_SOURCE_DIR}/src")
    set(BULLET_LIBRARIES BulletDynamics BulletCollision LinearMath)
endif()
find_package(TBB REQUIRED)
find_package(OpenVDB CONFIG QUIET)
if(NOT TARGET OpenVDB::openvdb)
    find_path(OPENVDB_INCLUDE_DIR openvdb/openvdb.h REQUIRED)
    find_library(OPENVDB_LIBRARY openvdb REQUIRED)
    find_package(Imath CONFIG QUIET)
    add_library(OpenVDB::openvdb UNKNOWN IMPORTED)
    set_target_properties(OpenVDB::openvdb PROPERTIES
        IMPORTED_LOCATION "${OPENVDB_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${OPENVDB_INCLUDE_DIR}")
    if(TARGET Imath::Imath)
        target_link_libraries(OpenVDB::openvdb INTERFACE Imath::Imath)
    else()
        find_library(ILMBASE_HALF_LIBRARY Half)
        if(ILMBASE_HALF_LIBRARY)
            target_link_libraries(OpenVDB::openvdb INTERFACE "${ILMBASE_HALF_LIBRARY}")
        endif()
    endif()
endif()

add_executable(voxel_world src/main.cpp src/terrain.cpp src/storage.cpp src/collision.cpp)
target_compile_features(voxel_world PRIVATE cxx_std_17)
target_include_directories(voxel_world SYSTEM PRIVATE ${BULLET_INCLUDE_DIRS})
target_link_libraries(voxel_world PRIVATE ${BULLET_LIBRARIES} OpenVDB::openvdb TBB::tbb)
if(MSVC)
    target_compile_options(voxel_world PRIVATE /W4)
else()
    target_compile_options(voxel_world PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
add_test(NAME voxel_world_correctness COMMAND voxel_world --self-test)
